cmake_minimum_required(VERSION 3.14)

project(tribol_install_consumer LANGUAGES CXX)

if(TRIBOL_CONSUMER_ENABLE_CUDA)
  enable_language(CUDA)
endif()

find_package(tribol CONFIG REQUIRED)

if(TRIBOL_CONSUMER_ENABLE_CUDA AND NOT TRIBOL_USE_CUDA)
  message(FATAL_ERROR "The install consumer expected a CUDA-enabled Tribol package")
endif()
if(TRIBOL_CONSUMER_ENABLE_HIP AND NOT TRIBOL_USE_HIP)
  message(FATAL_ERROR "The install consumer expected a HIP-enabled Tribol package")
endif()

add_executable(tribol_install_consumer consumer.cpp)
target_link_libraries(tribol_install_consumer PRIVATE tribol::core)
target_compile_features(tribol_install_consumer PRIVATE cxx_std_20)

if(NOT TRIBOL_USE_CUDA AND NOT TRIBOL_USE_HIP)
  add_executable(tribol_mfem_install_consumer mfem_consumer.cpp)
  target_link_libraries(tribol_mfem_install_consumer PRIVATE tribol::mfem_adapter)
  target_compile_features(tribol_mfem_install_consumer PRIVATE cxx_std_20)
endif()

if(TRIBOL_USE_CUDA OR TRIBOL_USE_HIP)
  if(NOT TARGET tribol::device_execution)
    message(FATAL_ERROR "A device-enabled Tribol package must export tribol::device_execution")
  endif()

  add_executable(tribol_device_install_consumer device_consumer.cpp)
  target_compile_features(tribol_device_install_consumer PRIVATE cxx_std_20)
  target_link_libraries(tribol_device_install_consumer PRIVATE tribol::device_execution)

  if(TRIBOL_USE_CUDA)
    if(NOT TARGET tribol::cuda_execution)
      message(FATAL_ERROR "A CUDA-enabled Tribol package must export tribol::cuda_execution")
    endif()
    set_source_files_properties(device_consumer.cpp PROPERTIES LANGUAGE CUDA)
    target_compile_features(tribol_device_install_consumer PRIVATE cuda_std_20)
    target_compile_definitions(tribol_device_install_consumer PRIVATE TRIBOL_CONSUMER_USE_CUDA)
    target_link_libraries(tribol_device_install_consumer PRIVATE tribol::cuda_execution)
  else()
    if(NOT TARGET tribol::hip_execution)
      message(FATAL_ERROR "A HIP-enabled Tribol package must export tribol::hip_execution")
    endif()
    target_compile_definitions(tribol_device_install_consumer PRIVATE TRIBOL_CONSUMER_USE_HIP)
    target_link_libraries(tribol_device_install_consumer PRIVATE tribol::hip_execution)
  endif()
endif()
